> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让飞牛音乐 App 在官方搜索框能搜到在线曲目并用 302 直链播放，同时提供安全的一键接管/还原能力。

**Architecture:** 新顶层包 `fnext/`（asyncio 单进程 FastAPI），容器内接管 `/var/run/trim_music.socket`：默认经 httpx UDS 全量透传到官方后端（改名后的 `trim_music_upstream.socket`），搜索/播放两个端点注入在线能力（复用 `app/search` + musicdl）。控制面走容器内 `127.0.0.1:8779` HTTP API + `fnextctl` CLI。与核心服务同镜像，compose profile `fnext` 启用。

**Tech Stack:** Python 3.11、FastAPI、uvicorn（UDS 监听）、httpx（UDS 透传）、musicdl（复用 `app.registry`/`app.search`）、pytest（离线单测，`asyncio.run` 驱动异步用例，不引入 pytest-asyncio）。

**范围说明：** 本计划只覆盖设计文档（`docs/superpowers/specs/2026-09-24-fnext-design.md`）的 **E1 里程碑**。E2（tee 边听边存）、E3（推荐）、E4（WebUI）各自在执行前另行出计划。

```python
    search_cache_neg_ttl_s: float = 30.0
    search_cache_empty_ttl_s: float = 10.0
    breaker_threshold: int = 4
    breaker_cooldown_s: float = 120.0
    slow_degrade_s: float = 8.0
    fast_return_items: int = 12
    slow_grace_s: float = 2.0
    intercept: bool = True
    record: bool = False
    record_path: str = "/app/fnext_data/recordings.jsonl"
    api_key: str | None = None

def _read_yaml(path: str) -> dict:
    try:
        with open(path, encoding="utf-8") as f:
            return yaml.safe_load(f) or {}
    except FileNotFoundError:
        return {}

def load(config_path: str = "config.yaml", overlay_path: str = "fnext.yaml") -> FnextConfig:
    cfg = FnextConfig()
    for k, v in _read_yaml(overlay_path).items():
        if hasattr(cfg, k):
            setattr(cfg, k, v)
    cfg.api_key = _read_yaml(config_path).get("api_key") or None
    for key in ("intercept", "record"):
        env = os.environ.get(f"FNEXT_{key.upper()}")
        if env in ("0", "1"):
            setattr(cfg, key, env == "1")
    return cfg
```

```python
# fnext/state.py
"""fnext 持久化状态：期望启停 + 原始 socket 元数据。原子写，防中途断电留半文件。"""
from __future__ import annotations
import json, os, time
from dataclasses import asdict, dataclass

@dataclass
class FnextState:
    desired: str = "enabled"
    socket_meta: dict | None = None
    updated_at: float = 0.0

class StateStore:
    def __init__(self, path: str):
        self.path = path

    def load(self) -> FnextState:
        try:
            with open(self.path, encoding="utf-8") as f:
                d = json.load(f)
            return FnextState(desired=d.get("desired", "enabled"),
                              socket_meta=d.get("socket_meta"),
                              updated_at=d.get("updated_at", 0.0))
        except (FileNotFoundError, json.JSONDecodeError, TypeError):
            return FnextState()

    def save(self, state: FnextState) -> None:
        state.updated_at = time.time()
        os.makedirs(os.path.dirname(self.path) or ".", exist_ok=True)
        part = self.path + ".part"
        with open(part, "w", encoding="utf-8") as f:
            json.dump(asdict(state), f, ensure_ascii=False, indent=2)
        os.replace(part, self.path)
```

**Interfaces:**
- Consumes: `StateStore`/`FnextState`（Task 1）
- Produces:
  - `TakeoverError(RuntimeError)`
  - `TakeoverManager(socket_path: str, upstream_path: str, store: StateStore)`
    - `.enable(wait_s: float = 30.0) -> None`（失败抛 `TakeoverError`；幂等）
    - `.disable() -> None`（幂等；拒绝删除非官方态 socket）
    - `.apply_perms() -> None`（按 `socket_meta` 对 `socket_path` 做 chown/chmod，供代理绑定后调用）
    - `.status() -> dict`（`{"desired","taken_over","socket_exists","upstream_exists"}`）

行为定义（测试即文档）：
- `enable`：已接管（upstream 存在且官方路径不在或是我们自己的残留）→ 直接返回；等待 `socket_path` 出现（轮询至 `wait_s`）；`upstream_path` 已存在且状态文件无 meta → 抛 `TakeoverError`（现场不明，拒绝）；记录 `st_uid/st_gid/st_mode` → `os.rename(socket_path, upstream_path)` → 保存状态（desired=enabled + meta）。
- `disable`：`upstream_path` 存在且 `socket_path` 不存在 → rename 回，清 meta，desired 保持；`upstream_path` 不存在 → 已是官方态，直接返回；两者都在 → 抛 `TakeoverError`（拒绝猜删）。

```python
# fnext/takeover.py
"""接管/还原状态机：只动 trim_music*.socket 两个路径；一切分支幂等；现场不明一律拒绝。"""
from __future__ import annotations
import os, stat, time

from .state import FnextState, StateStore

class TakeoverError(RuntimeError):
    pass

class TakeoverManager:
    def __init__(self, socket_path: str, upstream_path: str, store: StateStore):
        self.socket_path = socket_path
        self.upstream_path = upstream_path
        self.store = store

    @staticmethod
    def _is_socket(path: str) -> bool:
        try:
            return stat.S_ISSOCK(os.stat(path).st_mode)
        except FileNotFoundError:
            return False

    def enable(self, wait_s: float = 30.0) -> None:
        st = self.store.load()
        if os.path.exists(self.upstream_path) and st.socket_meta:
            st.desired = "enabled"
            self.store.save(st)
            return  # 已接管（如崩溃重启），直接恢复期望态
        if os.path.exists(self.upstream_path) and not st.socket_meta:
            raise TakeoverError("refuse: upstream socket exists but no state meta; keep scene intact")
        deadline = time.time() + wait_s
        while not self._is_socket(self.socket_path):
            if time.time() >= deadline:
                raise TakeoverError(f"timeout: official socket not found after {wait_s}s")
            time.sleep(0.2)
        meta = os.stat(self.socket_path)
        os.rename(self.socket_path, self.upstream_path)
        self.store.save(FnextState(
            desired="enabled",
            socket_meta={"uid": meta.st_uid, "gid": meta.st_gid,
                         "mode": stat.S_IMODE(meta.st_mode)},
            updated_at=0.0))

    def apply_perms(self) -> None:
        st = self.store.load()
        if not st.socket_meta or not os.path.exists(self.socket_path):
            return
        os.chown(self.socket_path, st.socket_meta["uid"], st.socket_meta["gid"])
        os.chmod(self.socket_path, st.socket_meta["mode"])

    def disable(self) -> None:
        up_exists = os.path.exists(self.upstream_path)
        sp_exists = os.path.exists(self.socket_path)
        if up_exists and sp_exists:
            raise TakeoverError("refuse: both exist (official recreated?); keep scene intact")
        st = self.store.load()
        if up_exists and not sp_exists:
            os.rename(self.upstream_path, self.socket_path)
        st.desired = "disabled"
        st.socket_meta = None
        self.store.save(st)

    def status(self) -> dict:
        return {
            "desired": self.store.load().desired,
            "taken_over": os.path.exists(self.upstream_path),
            "socket_exists": os.path.exists(self.socket_path),
            "upstream_exists": os.path.exists(self.upstream_path),
        }
```

```python
class ServerRunner:
    """uvicorn 生命周期封装：enable/disable 状态机需要反复启停代理 server。"""
    def __init__(self, app, uds: str | None = None, host: str | None = None, port: int | None = None):
        config = uvicorn.Config(app, uds=uds, host=host, port=port, log_level="info")
        self.server = uvicorn.Server(config)
        self._task: asyncio.Task | None = None

    async def start(self, timeout_s: float = 10.0) -> None:
        self._task = asyncio.create_task(self.server.serve())
        for _ in range(int(timeout_s * 20)):
            if self.server.started:
                return
            await asyncio.sleep(0.05)
        raise RuntimeError("server start timeout")

    async def stop(self) -> None:
        self.server.should_exit = True
        if self._task:
            await self._task
            self._task = None

def _clean_headers(headers, drop_host=True):
    return {k: v for k, v in headers.items()
            if k.lower() not in HOP_BY_HOP and (not drop_host or k.lower() != "host")}

async def forward_json(client: httpx.AsyncClient, request: Request, path: str) -> dict:
    """透传请求并解析官方 JSON 响应；供拦截器拿本地结果。"""
    body = await request.body()
    resp = await client.request(
        request.method, path, params=dict(request.query_params), content=body,
        headers=_clean_headers(request.headers))
    return resp.json()

def create_app(*, cfg: FnextConfig, takeover: TakeoverManager | None,
               upstream_client_factory=None,
               search_handler=None, stream_handler=None) -> FastAPI:
    app = FastAPI(title="fnext")
    client = upstream_client_factory() if upstream_client_factory else httpx.AsyncClient(
        transport=httpx.AsyncHTTPTransport(uds=cfg.upstream_path), timeout=30.0)
    record_lock = asyncio.Lock()

    async def _record(request: Request, status: int, resp_headers, resp_body: bytes):
        entry = {
            "ts": time.time(),
            "method": request.method,
            "path": request.url.path,
            "query": str(request.query_params),
            "req_headers": dict(request.headers),
            "req_body_b64": base64.b64encode(await request.body()).decode(),
            "status": status,
            "resp_headers": dict(resp_headers),
            "resp_body_b64": base64.b64encode(resp_body).decode(),
        }
        async with record_lock:
            with open(cfg.record_path, "a", encoding="utf-8") as f:
                f.write(json.dumps(entry, ensure_ascii=False) + "\n")

    @app.get("/_ext/healthz")
    async def healthz():
        upstream_ok = False
        try:
            r = await client.get(SEARCH_PATH, params={"keyword": "healthz_probe"}, timeout=2.0)
            upstream_ok = r.status_code < 500
        except Exception:
            pass
        return {"ok": upstream_ok, "upstream": upstream_ok,
                "takeover": takeover.status() if takeover else None, "version": _VERSION}

    if cfg.intercept and search_handler:
        app.add_api_route(SEARCH_PATH, search_handler, methods=["GET"])
    if cfg.intercept and stream_handler:
        app.add_api_route(STREAM_PATH, stream_handler, methods=["GET"])

    @app.api_route("/{path:path}", methods=["GET", "POST", "PUT", "DELETE", "PATCH", "HEAD", "OPTIONS"])
    async def passthrough(request: Request, path: str):
        body = await request.body()
        req = client.build_request(request.method, f"/{path}",
                                   params=dict(request.query_params), content=body,
                                   headers=_clean_headers(request.headers))
        resp = await client.send(req, stream=True)
        out_headers = _clean_headers(resp.headers, drop_host=False)
        ctype = resp.headers.get("content-type", "")
        textual = "json" in ctype or ctype.startswith("text/")
        if cfg.record and textual and resp.status_code < 500:
            content = await resp.aread()
            if len(content) <= 1024 * 1024:
                await _record(request, resp.status_code, resp.headers, content)
            await resp.aclose()
            return Response(content=content, status_code=resp.status_code, headers=out_headers)
        return StreamingResponse(resp.aiter_bytes(), status_code=resp.status_code,
                                 headers=out_headers,
                                 background=None)
    return app
```

```python
class FnextDaemon:
    def __init__(self, cfg: FnextConfig, store: StateStore | None, takeover: TakeoverManager,
                 search_handler=None, stream_handler=None):
        self.cfg = cfg
        self.store = store
        self.takeover = takeover
        self.search_handler = search_handler
        self.stream_handler = stream_handler
        self.runner: ServerRunner | None = None
        self.control_runner: ServerRunner | None = None

    async def enable(self) -> None:
        self.takeover.enable(wait_s=30.0)
        app = create_app(cfg=self.cfg, takeover=self.takeover,
                         search_handler=self.search_handler,
                         stream_handler=self.stream_handler)
        self.runner = ServerRunner(app, uds=self.cfg.socket_path)
        await self.runner.start()
        self.takeover.apply_perms()

    async def disable(self) -> None:
        if self.runner:
            await self.runner.stop()
            self.runner = None
        self.takeover.disable()

    def status(self) -> dict:
        return {"takeover": self.takeover.status(),
                "proxy_running": self.runner is not None}

    async def run(self) -> int:
        self.control_runner = ServerRunner(create_control_app(self),
                                           host=self.cfg.control_host,
                                           port=self.cfg.control_port)
        await self.control_runner.start()
        taken_over = False
        if self.store is None or self.store.load().desired == "enabled":
            await self.enable()
            taken_over = True
        stop = asyncio.Event()
        loop = asyncio.get_running_loop()
        for sig in (signal.SIGTERM, signal.SIGINT):
            loop.add_signal_handler(sig, stop.set)
        await stop.wait()
        if taken_over:
            try:
                await self.disable()
            except Exception:
                pass  # 退出路径不抛错，看门狗兜底
        if self.control_runner:
            await self.control_runner.stop()
        return 0
```

```python
# fnext/online.py
"""在线引擎原语：每源舱壁 / 自适应超时 / 熔断 / 分层搜索缓存。纯标准库，线程安全。
机制借鉴 javycoder/fnos_music_ext 的 musicdl-service/hardening.py（MIT）。"""
from __future__ import annotations
import threading, time
from collections import OrderedDict
from concurrent.futures import ThreadPoolExecutor
from dataclasses import dataclass, field

class SourceBusy(Exception):
    pass

class SourceBulkhead:
    """每源一个 worker 线程 + 信号量准入：忙即拒，不排队。"""
    def __init__(self):
        self._pools: dict[str, ThreadPoolExecutor] = {}
        self._permits: dict[str, threading.Semaphore] = {}
        self._lock = threading.Lock()

    def submit(self, source: str, fn, *args, **kwargs):
        with self._lock:
            pool = self._pools.setdefault(
                source, ThreadPoolExecutor(max_workers=1, thread_name_prefix=f"src-{source}"))
            permit = self._permits.setdefault(source, threading.Semaphore(1))
        if not permit.acquire(blocking=False):
            raise SourceBusy(source)
        fut = pool.submit(fn, *args, **kwargs)
        fut.add_done_callback(lambda f: permit.release())
        return fut

class AdaptiveTimeout:
    """失败 ×0.6 收紧（下限 min），成功 ×1.25 恢复（上限 base）。"""
    def __init__(self, base: float = 12.0, min_timeout: float = 8.0):
        self.base, self.min = base, min_timeout
        self._t: dict[str, float] = {}
        self._lock = threading.Lock()

    def timeout_for(self, source: str) -> float:
        with self._lock:
            return self._t.get(source, self.base)

    def record_failure(self, source: str) -> None:
        with self._lock:
            cur = self._t.get(source, self.base)
            self._t[source] = max(self.min, cur * 0.6)

    def record_success(self, source: str) -> None:
        with self._lock:
            cur = self._t.get(source, self.base)
            self._t[source] = min(self.base, cur * 1.25)

class SourceBreaker:
    """连败 threshold 次熔断 cooldown 秒；冷却结束自动半开放行；慢成功（>slow_s）也计失败。"""
    def __init__(self, threshold: int = 4, cooldown_s: float = 120.0):
        self.threshold, self.cooldown = threshold, cooldown_s
        self._fails: dict[str, int] = {}
        self._opened: dict[str, float] = {}
        self._lock = threading.Lock()

    def is_open(self, source: str) -> bool:
        with self._lock:
            opened = self._opened.get(source)
            if opened is None:
                return False
            if time.time() - opened >= self.cooldown:
                self._opened.pop(source, None)
                self._fails[source] = 0
                return False
            return True

    def record_failure(self, source: str) -> None:
        with self._lock:
            n = self._fails.get(source, 0) + 1
            self._fails[source] = n
            if n >= self.threshold:
                self._opened[source] = time.time()

    def record_success(self, source: str, latency_s: float | None = None, slow_s: float = 8.0) -> None:
        if latency_s is not None and latency_s > slow_s:
            self.record_failure(source)
            return
        with self._lock:
            self._fails[source] = 0
            self._opened.pop(source, None)

@dataclass
class CacheEntry:
    items: list
    errors: dict
    expires: float

class SearchCache:
    """关键词级 LRU+TTL。分层：全成功 ttl / 有结果但有错误 neg_ttl / 空结果 empty_ttl。"""
    def __init__(self, ttl: float = 300.0, neg_ttl: float = 30.0,
                 empty_ttl: float = 10.0, max_entries: int = 200):
        self.ttl, self.neg_ttl, self.empty_ttl = ttl, neg_ttl, empty_ttl
        self.max_entries = max_entries
        self._entries: OrderedDict[tuple, CacheEntry] = OrderedDict()
        self._lock = threading.Lock()

    @staticmethod
    def _key(keyword: str, sources: list[str], limit: int):
        return (keyword.strip(), "|".join(sorted(sources)), limit)

    def get(self, keyword: str, sources: list[str], limit: int, alive=None):
        with self._lock:
            e = self._entries.get(self._key(keyword, sources, limit))
            if not e:
                return None
            if time.time() > e.expires:
                del self._entries[self._key(keyword, sources, limit)]
                return None
            if alive and e.items and not all(alive(getattr(t, "guid", t)) for t in e.items):
                del self._entries[self._key(keyword, sources, limit)]
                return None
            self._entries.move_to_end(self._key(keyword, sources, limit))
            return e.items

    def put(self, keyword: str, sources: list[str], limit: int, items: list, errors: dict) -> None:
        life = self.ttl if not errors else self.neg_ttl
        if not items:
            life = self.empty_ttl
        with self._lock:
            self._entries[self._key(keyword, sources, limit)] = CacheEntry(
                items=list(items), errors=dict(errors), expires=time.time() + life)
            while len(self._entries) > self.max_entries:
                self._entries.popitem(last=False)
```

```python
# ---- 以下追加到 fnext/online.py ----
import asyncio
from dataclasses import dataclass, field

@dataclass
class OnlineTrack:
    guid: str
    source: str
    identifier: str
    title: str
    artists: list[str]
    album: str | None
    duration_s: int | None
    cover_url: str | None
    lyric: str | None
    download_url: str | None
    keyword: str
    raw: dict
    alternatives: list[dict] = field(default_factory=list)

def guid_encode(source: str, identifier: str) -> str:
    return f"online:{source}:{identifier}"

def guid_decode(guid: str) -> tuple[str, str]:
    parts = guid.split(":", 2)
    if len(parts) != 3 or parts[0] != "online":
        raise ValueError(f"not an online guid: {guid!r}")
    return parts[1], parts[2]

class SongCache:
    """曲目级缓存：播放解析只凭 guid；超限淘汰最旧一半。"""
    def __init__(self, ttl: float = 1800.0, max_entries: int = 3000):
        self.ttl, self.max_entries = ttl, max_entries
        self._d: OrderedDict[str, tuple[OnlineTrack, float]] = OrderedDict()
        self._lock = threading.Lock()

    def put(self, track: OnlineTrack) -> None:
        with self._lock:
            self._d[track.guid] = (track, time.time())
            self._d.move_to_end(track.guid)
            if len(self._d) > self.max_entries:
                for k in list(self._d.keys())[: len(self._d) // 2]:
                    del self._d[k]

    def get(self, guid: str) -> OnlineTrack | None:
        with self._lock:
            item = self._d.get(guid)
            if not item:
                return None
            t, ts = item
            if time.time() - ts > self.ttl:
                del self._d[guid]
                return None
            return t

    def has(self, guid: str) -> bool:
        return self.get(guid) is not None

def _same_recording(a: OnlineTrack, b: OnlineTrack) -> bool:
    if a.title.casefold() != b.title.casefold():
        return False
    if ",".join(a.artists).casefold() != ",".join(b.artists).casefold():
        return False
    if a.duration_s and b.duration_s and abs(a.duration_s - b.duration_s) > 2:
        return False
    return True

def dedup_merge(tracks: list[OnlineTrack]) -> list[OnlineTrack]:
    by_guid: dict[str, OnlineTrack] = {}
    for t in tracks:
        by_guid.setdefault(t.guid, t)
    out: list[OnlineTrack] = []
    for t in by_guid.values():
        dup = next((o for o in out if _same_recording(o, t)), None)
        if dup:
            dup.alternatives.append({"source": t.source, "identifier": t.identifier,
                                     "download_url": t.download_url})
        else:
            out.append(t)
    return out

def musicdl_search_fn(source: str, keyword: str, limit: int) -> list[OnlineTrack]:
    """真实适配器：单源调用 musicdl 并标准化。同步阻塞，由舱壁线程执行。"""
    from app.registry import build_client
    from app.search import normalize_song
    client = build_client([source])
    results = client.search(keyword=keyword) or {}
    out: list[OnlineTrack] = []
    for s, songs in results.items():
        for song in songs or []:
            t = normalize_song(s, song)
            if not t or not (t.raw or {}).get("download_url"):
                continue
            identifier = t.id.split(":", 1)[1] if ":" in t.id else t.id
            out.append(OnlineTrack(
                guid=guid_encode(s, identifier), source=s, identifier=identifier,
                title=t.title, artists=list(t.artists), album=t.album,
                duration_s=t.duration_s, cover_url=t.cover_url, lyric=t.lyric,
                download_url=(t.raw or {}).get("download_url"), keyword=keyword,
                raw=t.raw or {}))
            if len(out) >= limit:
                return out
    return out

class OnlineEngine:
    def __init__(self, cfg):
        self.cfg = cfg
        self.bulkhead = SourceBulkhead()
        self.adaptive = AdaptiveTimeout(cfg.search_timeout_s, cfg.search_min_timeout_s)
        self.breaker = SourceBreaker(cfg.breaker_threshold, cfg.breaker_cooldown_s)
        self.cache = SearchCache(cfg.search_cache_ttl_s, cfg.search_cache_neg_ttl_s,
                                 cfg.search_cache_empty_ttl_s)
        self.songs = SongCache()

    async def search(self, keyword: str, limit: int, search_fn=None):
        search_fn = search_fn or musicdl_search_fn
        sources = list(self.cfg.online_sources)
        cached = self.cache.get(keyword, sources, limit, alive=self.songs.has)
        if cached is not None:
            return cached[:limit], {}
        errors: dict[str, str] = {}
        loop = asyncio.get_running_loop()

        async def run_source(src):
            if self.breaker.is_open(src):
                errors[src] = "circuit breaker open"
                return []
            try:
                fut = self.bulkhead.submit(src, search_fn, src, keyword, limit)
            except SourceBusy:
                errors[src] = "busy"
                return []
            timeout = self.adaptive.timeout_for(src)
            start = loop.time()
            try:
                items = await asyncio.wait_for(asyncio.wrap_future(fut), timeout=timeout)
            except asyncio.TimeoutError:
                errors[src] = f"timeout after {timeout:.1f}s"
                self.adaptive.record_failure(src)
                self.breaker.record_failure(src)
                return []
            except Exception as e:
                errors[src] = f"{type(e).__name__}: {e}"
                self.adaptive.record_failure(src)
                self.breaker.record_failure(src)
                return []
            latency = loop.time() - start
            self.adaptive.record_success(src)
            self.breaker.record_success(src, latency, self.cfg.slow_degrade_s)
            return items

        tasks = {asyncio.create_task(run_source(s)): s for s in sources}
        results: list[OnlineTrack] = []
        deadline = loop.time() + max(self.cfg.search_timeout_s, 4.0)
        fast = False
        while tasks:
            remaining = deadline - loop.time()
            if remaining <= 0:
                break
            done, _ = await asyncio.wait(set(tasks), timeout=remaining,
                                         return_when=asyncio.FIRST_COMPLETED)
            for t in done:
                src = tasks.pop(t)
                results.extend(t.result())
            if len(results) >= self.cfg.fast_return_items:
                fast = True
                break
            if results:
                deadline = min(deadline, loop.time() + self.cfg.slow_grace_s)
        for t, src in tasks.items():
            t.cancel()
            errors.setdefault(src, "skipped (fast return)" if fast
                              else "timeout (partial results)")
        uniq = dedup_merge(results)
        for t in uniq:
            self.songs.put(t)
        self.cache.put(keyword, sources, limit, uniq, errors)
        return uniq[:limit], errors
```

```python
"""录制 JSONL → 契约 fixtures：只保留 json/text 且 ≤1MB 的响应；body 解码为文本；剔除敏感头。"""
from __future__ import annotations
import base64, json, os, re, sys

SENSITIVE = {"cookie", "authorization", "x-api-key", "set-cookie"}
MAX_BODY = 1024 * 1024

def _scrub(headers: dict) -> dict:
    return {k: v for k, v in headers.items() if k.lower() not in SENSITIVE}

def _name(rec: dict, seen: set) -> str:
    base = f"{rec['method']}_{rec['path'].strip('/').replace('/', '_') or 'root'}"
    base = re.sub(r"[^A-Za-z0-9_.-]", "_", base)
    name, i = base, 2
    while name in seen:
        name, i = f"{base}_{i}", i + 1
    seen.add(name)
    return name + ".json"

def distill(records_path: str, out_dir: str) -> list[str]:
    os.makedirs(out_dir, exist_ok=True)
    written, seen = [], set()
    with open(records_path, encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            rec = json.loads(line)
            ctype = rec.get("resp_headers", {}).get("content-type", "")
            if "json" not in ctype and not ctype.startswith("text/"):
                continue
            body = base64.b64decode(rec.get("resp_body_b64") or b"")
            if len(body) > MAX_BODY:
                continue
            fixture = {
                "request": {"method": rec["method"], "path": rec["path"],
                            "query": rec.get("query", ""),
                            "headers": _scrub(rec.get("req_headers", {}))},
                "response": {"status": rec["status"],
                             "headers": _scrub(rec.get("resp_headers", {})),
                             "body": body.decode("utf-8", errors="replace")},
            }
            name = _name(rec, seen)
            with open(os.path.join(out_dir, name), "w", encoding="utf-8") as out:
                json.dump(fixture, out, ensure_ascii=False, indent=2)
            written.append(name)
    return written

if __name__ == "__main__":
    print("\n".join(distill(sys.argv[1], sys.argv[2])))
```

```python
# fnext/intercept.py
"""搜索/播放拦截器：本地官方结果优先，在线结果按官方 track 模板合并追加。"""
from __future__ import annotations
import json
from fastapi.responses import JSONResponse, RedirectResponse

from .online import OnlineEngine, OnlineTrack, guid_decode

_LIST_KEYS = ("tracks", "list", "items", "songs", "result")
_GUID_KEYS = ("guid", "id", "songId", "trackId")
_TITLE_KEYS = ("songname", "title", "name")
_ARTIST_KEYS = ("singer", "artist", "artists", "singername")
_ALBUM_KEYS = ("album", "albumname", "album_name")
_DURATION_KEYS = ("duration", "interval", "duration_s")
_COVER_KEYS = ("cover", "coverUrl", "cover_url", "pic", "img")

def _pick_key(d: dict, candidates: tuple[str, ...]) -> str | None:
    return next((k for k in candidates if k in d), None)

def _find_track_list(obj) -> list:
    """在官方响应里找曲目列表：data/result 下一层的 tracks/list/items/songs。"""
    if isinstance(obj, dict):
        for wrap in ("data", "result", obj):
            node = obj.get(wrap) if isinstance(wrap, str) else obj
            if isinstance(node, dict):
                for k in _LIST_KEYS:
                    if isinstance(node.get(k), list):
                        return node[k]
            if isinstance(node, list):
                return node
    return []

def build_online_entry(ot: OnlineTrack, template: dict) -> dict:
    entry = json.loads(json.dumps(template))  # 以官方模板为底，保留未知字段
    def put(candidates, value):
        k = _pick_key(entry, candidates)
        if k is not None and value is not None:
            entry[k] = value
    put(_GUID_KEYS, ot.guid)
    put(_TITLE_KEYS, ot.title)
    put(_ARTIST_KEYS, "/".join(ot.artists) if isinstance(
        entry.get(_pick_key(entry, _ARTIST_KEYS) or ""), str) else ot.artists)
    put(_ALBUM_KEYS, ot.album)
    put(_DURATION_KEYS, ot.duration_s)
    put(_COVER_KEYS, ot.cover_url)
    return entry

def merge_online_into_official(official: dict, tracks: list[OnlineTrack]) -> dict:
    lst = _find_track_list(official)
    if not lst:
        return official  # 结构不明时不冒险，原样返回
    template = lst[0] if lst else None
    for ot in tracks:
        lst.append(build_online_entry(ot, template) if template else {"guid": ot.guid})
    return official

def make_search_handler(client, engine: OnlineEngine, cfg):
    async def handler(request):
        keyword = request.query_params.get("keyword", "")
        official = {}
        try:
            resp = await client.request("GET", "/music/api/v1/search/track",
                                        params=dict(request.query_params),
                                        headers={k: v for k, v in request.headers.items()
                                                 if k.lower() not in ("host", "connection")})
            official = resp.json()
        except Exception:
            pass  # 官方异常不挡在线结果
        try:
            limit = int(request.query_params.get("limit", 20))
        except ValueError:
            limit = 20
        tracks, _errors = await engine.search(keyword, limit)
        return JSONResponse(merge_online_into_official(official, tracks))
    return handler

def make_stream_handler(client, engine: OnlineEngine, passthrough):
    """E1 播放：online guid → 302 直链；本地 guid → 透传；失忆 → 404 提示重新搜索。"""
    async def handler(request):
        guid = request.query_params.get("guid") or request.query_params.get("id") or ""
        try:
            guid_decode(guid)
        except ValueError:
            return await passthrough(request)
        track = engine.songs.get(guid)
        if track is None:
            return JSONResponse({"code": 4040, "msg": "online track expired, search again"},
                                status_code=404)
        url = track.download_url
        if not url:
            for alt in track.alternatives:
                if alt.get("download_url"):
                    url = alt["download_url"]
                    break
        if not url:
            return JSONResponse({"code": 4041, "msg": "no playable url"}, status_code=404)
        return RedirectResponse(url, status_code=302)
    return handler
```
